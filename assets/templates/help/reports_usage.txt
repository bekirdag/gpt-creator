  gpt-creator reports auto [--project PATH] [--reporter NAME] [--no-push] [--prompt-only]

  gpt-creator reports [--project PATH] [--open] <slug>

                               [--label-invalid NAME|--no-label-invalid]
                               [--comment TEXT]
